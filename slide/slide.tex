\section{Problem and Goal}
% ============================================================

\begin{frame}{Problem Definition and Research Goal}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{Fractal Image Compression (FIC)}
        FIC represents an image by self-similar block mappings:
        a small \alert{range block} is approximated by a transformed larger
        \alert{domain block}.
      \end{block}
      \begin{alertblock}{Encoding bottleneck}
        For each range block, full search evaluates every domain block and all 8
        isometries:
        \[
          M\times P\times 8
          =65{,}536\times65{,}025\times8
          \approx \mathbf{3.4\times10^{10}}
        \]
        candidate evaluations per $1024\times1024$ image.
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.9]
        \draw[step=0.4cm, gray!50, thin] (0,0) grid (3.2,3.2);

\section{Encoding and Decoding}
% ============================================================

\begin{frame}{Solution Representation: How FIC Encodes and Decodes}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{One fractal code per range block}
        \[
          C_i=(d^*, k^*, s^*, o^*)
        \]
        \begin{itemize}
          \item $d^*$: selected domain block index
          \item $k^*$: one of 8 rotations/flips
          \item $s^*, o^*$: contrast and brightness
        \end{itemize}
      \end{block}
      \[
        \hat v = s\cdot T_k(\text{downsample}(u_d)) + o
      \]
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Fitness function evaluation (FFE)}
        For a fixed $(d,k)$, $s$ and $o$ have a closed-form least-squares solution:
        \[
          s=\frac{\mathrm{Cov}(v,u_k)}{\mathrm{Var}(u_k)},\qquad
          o=\bar v-s\bar u_k.
        \]
        One MSE computation for one $(d,k)$ pair is one \alert{FFE}.
      \end{block}
      \begin{exampleblock}{Decoding}
        Start from any image and repeatedly apply all stored maps. Because maps are
        contractive ($|s|<1$), the process converges to a reconstruction in about
        20 iterations.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Proposed Solution Flow}
% ============================================================

\begin{frame}{Why PSO Alone Is Not Enough}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Prior idea}
        Treat $(d,k)$ as the decision variables and search using PSO or GA.
      \end{block}
      \begin{alertblock}{High-resolution issue}
        One range block has
        \[
          P\times8 = 65{,}025\times8 = 520{,}200
        \]
        candidates. A 40-particle swarm samples only a tiny fraction, and the
        domain-index objective is non-smooth.
      \end{alertblock}
    \end{column}
    \begin{column}{0.52\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.86, transform shape, node distance=0.45cm and 0.7cm]
        \node[box] (pso) {\textbf{PSO}\\ full pool};
        \node[box, right=of pso] (mpso) {\textbf{Memetic PSO}\\ PSO + LS};
        \node[box, right=of mpso] (fgpso) {\textbf{FG-PSO}\\ feature pool + PSO};
        \node[greenbox, below=0.7cm of fgpso] (fgds) {\textbf{FGDS}\\ feature pool + exhaustive};
        \draw[flow] (pso)--(mpso);
        \draw[flow] (mpso)--(fgpso);
        \draw[flow] (fgpso)--(fgds);
      \end{tikzpicture}
      \vspace{0.25cm}
      \begin{exampleblock}{Key ablation}
        Once the pool is small, \textbf{exhaustive search beats PSO inside it}.
        Therefore the useful heuristic is not the inner optimizer, but the way we
        construct the candidate pool.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Our Method: Feature-Guided Domain Selection (FGDS)}
  \centering
  \begin{tikzpicture}[node distance=0.35cm and 0.55cm]
    \node[bluebox] (img) {Image\\ blocks};
    \node[bluebox, right=of img] (feat) {Feature\\ extraction\\ 8-D};
    \node[bluebox, right=of feat] (wfs) {Weighted\\ feature\\ space};
    \node[box, right=of wfs] (kd) {KD-Tree\\ KNN};
    \node[box, right=of kd] (pool) {Candidate\\ pool};
    \node[greenbox, right=of pool] (exh) {Exhaustive\\ search};
    \draw[flow] (img)--(feat);
    \draw[flow] (feat)--(wfs);
    \draw[flow] (wfs)--(kd);
    \draw[flow] (kd)--(pool);
    \draw[flow] (pool)--(exh);
    \node[below=0.55cm of wfs, font=\scriptsize, accentorange, text width=3.2cm, align=center]
        (ga) {GA learns feature weights\\ offline once};
    \draw[-{Stealth}, thick, accentorange, dashed] (ga) -- (wfs);
  \end{tikzpicture}

  \vspace{0.35cm}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Online encoding}
        \begin{enumerate}
          \item Extract features for range/domain blocks.
          \item Query KD-Tree for top-$K$ similar domains.
          \item Try all $K\times8$ candidates and store the best code.
        \end{enumerate}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{At $K=80$}
        Per-range search space:
        \[
          520{,}200 \rightarrow 640
        \]
        over \textbf{800$\times$ fewer} candidates, yet exhaustive within the selected pool.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Feature Extraction and Candidate Selection}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{8-D isometry-invariant descriptor}
      \[
        f(b)=
        [\mu,\sigma,\bar g_h,\bar g_v,\hat q_1,\hat q_2,\hat q_3,\hat q_4]
      \]
      \begin{itemize}
        \item $\mu,\sigma$: intensity and contrast.
        \item $\bar g_h,\bar g_v$: texture via gradient magnitudes.
        \item $\hat q_{1..4}$: sorted quadrant means, invariant to rotations/flips.
      \end{itemize}
      \begin{block}{Distance}
        \[
          \|\mathbf w\odot(\hat f_R-\hat f_D)\|^2
        \]
        Features are normalized; $\mathbf w$ is learned offline.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{KD-Tree candidate selection}
        Pairwise feature search:
        \[
          O(MPF)\approx179.6\text{ s}
        \]
        KD-Tree:
        \[
          O(PF\log P+MK\log P)\approx0.13\text{ s}
        \]
        \centering \alert{about 1370$\times$ faster, same nearest neighbours.}
      \end{exampleblock}
      \begin{alertblock}{Why features matter}
        A random pool with the same $K$ performs much worse; the quality comes from
        feature-guided selection, not just from using a smaller pool.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{What Is a KD-Tree?}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Idea}
        A \alert{KD-Tree} stores multi-dimensional points by recursively splitting
        the feature space.
        \begin{itemize}
          \item Each node chooses one feature dimension and splits by a median value.
          \item A query descends to nearby regions first.
          \item Branches that cannot contain closer points are pruned.
        \end{itemize}
      \end{block}
      \begin{exampleblock}{In FGDS}
        \begin{itemize}
          \item Points: 8-D features of all domain blocks.
          \item Query: 8-D feature of one range block.
          \item Output: top-$K$ nearest candidate domains.
        \end{itemize}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.52\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.95]
        \draw[thick] (0,0) rectangle (4.1,3.0);
        \draw[accentblue, very thick] (1.75,0)--(1.75,3.0);
        \node[accentblue, font=\tiny, above] at (1.75,3.0){split by $\sigma$};
        \draw[accentgreen, thick] (0,1.75)--(1.75,1.75);
        \draw[accentgreen, thick] (1.75,1.15)--(4.1,1.15);
        \node[accentgreen, font=\tiny, right] at (4.05,1.15){split by $q_1$};
        \foreach \p in {(0.55,2.35),(1.05,0.9),(0.85,2.65),(1.25,1.1),
                        (2.35,2.1),(3.1,0.6),(2.65,2.55),(3.0,1.85),(2.05,0.75)}
          \fill[black] \p circle (1.7pt);
        \fill[ntnured] (2.55,2.25) circle (2.8pt);
        \node[ntnured, font=\tiny, right] at (2.62,2.42){range query};
        \draw[ntnured, dashed, thick] (2.55,2.25) circle (0.72cm);
        \node[font=\scriptsize] at (2.05,-0.35){Only 2 of the 8 feature dimensions shown};
      \end{tikzpicture}
      \vspace{0.25cm}
      \begin{alertblock}{Why this helps}
        Instead of comparing one range block to all $65{,}025$ domains, the tree quickly
        narrows the search to nearby feature regions.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Genetic Algorithm: Learning Feature Weights}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Why GA?}
        Equal weights assume all features matter equally. But in the affine model,
        mean mismatch is absorbed by $o$, while contrast mismatch affects $s$ and MSE.
      \end{block}
      \begin{exampleblock}{GA setup}
        \begin{itemize}
          \item Individual: 8 feature weights.
          \item Fitness: negative average MSE on 2\% sampled range blocks.
          \item Population 16, 10 generations, $K=80$.
          \item Training image: \texttt{0003}, not in the test set.
        \end{itemize}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Learned weights}
      \[
        [0.18,\,1.36,\,0.97,\,0.96,\,1.29,\,1.04,\,1.00,\,0.75]
      \]
      \begin{itemize}
        \item Smallest: mean $\mu=0.18$.
        \item Largest: standard deviation $\sigma=1.36$.
        \item Ratio $\sigma/\mu\approx7.6$.
      \end{itemize}
      \begin{alertblock}{Role of GA}
        GA is \textbf{not} the main source of gain; it is a small but consistent refinement
        (+0.05 dB average), positive on all 10 test images.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{GA Crossover: How New Feature Weights Are Created}
  \small
  \begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.48\textwidth}
      We use real-valued chromosomes:
      \vspace{-0.1cm}
      \[
        \mathbf w=[w_\mu,w_\sigma,w_{g_h},w_{g_v},w_{q_1},...,w_{q_4}]
      \]
      \vspace{-0.18cm}
      For crossover, we use \alert{BLX-$\alpha$}:
      for each gene, given two parent values $p_1,p_2$,
      \vspace{-0.08cm}
      \[
        I=|p_2-p_1|,
      \]
      \vspace{-0.28cm}
      \[
      \begin{aligned}
        w_{\text{child}} &\sim U[L,H],\\
        L &= \min(p_1,p_2)-\alpha I,\\
        H &= \max(p_1,p_2)+\alpha I,\quad \alpha=0.5.
      \end{aligned}
      \]
      \vspace{-0.18cm}
      \begin{block}{After crossover}
        With probability 0.4, mutate 1--3 genes by Gaussian noise, then clamp
        weights to remain positive.
      \end{block}
    \end{column}
    \begin{column}{0.52\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.78, transform shape]
        \draw[->] (-0.3,0)--(6.2,0) node[right,font=\scriptsize]{weight};
        \fill[accentblue] (1.6,0) circle (2.8pt) node[above=2pt,font=\scriptsize,accentblue]{$p_1$};
        \fill[ntnured] (4.1,0) circle (2.8pt) node[above=2pt,font=\scriptsize,ntnured]{$p_2$};
        \draw[thick] (1.6,-0.12)--(1.6,0.12);
        \draw[thick] (4.1,-0.12)--(4.1,0.12);
        \draw[accentgreen, very thick] (0.35,-0.35)--(5.35,-0.35);
        \draw[accentgreen] (0.35,-0.5)--(0.35,-0.2);
        \draw[accentgreen] (5.35,-0.5)--(5.35,-0.2);
        \draw[decorate, decoration={brace, amplitude=4pt}] (1.6,0.45)--(4.1,0.45);
        \node[font=\scriptsize] at (2.85,0.78) {$I=|p_2-p_1|$};
        \node[accentgreen, font=\scriptsize] at (2.85,-0.72)
          {extended sampling interval};
        \fill[black] (3.05,0) circle (2.8pt) node[below=4pt,font=\scriptsize]{child};
        \draw[decorate, decoration={brace, amplitude=3pt}] (0.35,-1.05)--(1.6,-1.05);
        \node[font=\tiny] at (0.98,-1.3){$\alpha I$};
        \draw[decorate, decoration={brace, amplitude=3pt}] (4.1,-1.05)--(5.35,-1.05);
        \node[font=\tiny] at (4.72,-1.3){$\alpha I$};
      \end{tikzpicture}
      \vspace{0.16cm}
      \begin{exampleblock}{Why BLX-$\alpha$}
        It samples between or slightly outside the parents, so GA can exploit good
        weights while exploring new importance patterns.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Conclusion}
% ============================================================

\begin{frame}{Conclusion and Future Work}
  \begin{block}{Conclusion}
    \begin{itemize}
      \item For high-resolution FIC, the key bottleneck is \textbf{search-space quality},
            not the inner optimizer.
      \item Feature-guided candidate selection + exhaustive search outperforms PSO-based
            matching and removes stochastic variance.
      \item KD-Tree makes candidate selection practical: $179.6$ s $\to 0.13$ s.
      \item GA-learned feature weights give a small but consistent improvement.
    \end{itemize}
  \end{block}
  \begin{exampleblock}{Future work}
    \begin{itemize}
      \item Extend to color images and different bit-rate settings.
      \item Adaptive $K$ based on local block complexity.
      \item Replace hand-crafted descriptors with learned embeddings.
    \end{itemize}
  \end{exampleblock}
\end{frame}
